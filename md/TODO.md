def _normalize_scores(self, scores: list, key: str) -> list:
    values = [item[key] for item in scores]  # берём значения по ключу
    min_val = min(values)
    max_val = max(values)
    
    for item in scores:
        item[f"{key}_norm"] = (item[key] - min_val) / (max_val - min_val)
    
    return scores
